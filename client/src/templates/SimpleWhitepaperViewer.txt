export const WHITEPAPER_TEMPLATE = `import React, { useState, useEffect } from 'react';

export default function App() {
  const [searchQuery, setSearchQuery] = useState('');
  const [showToc, setShowToc] = useState(true);
  const [activeSection, setActiveSection] = useState('');

  // Whitepaper content - will be injected by AI
  const content = \`{{WHITEPAPER_CONTENT}}\`;
  const title = "{{TITLE}}";
  const subtitle = "{{SUBTITLE}}";
  const author = "{{AUTHOR}}";
  const date = "{{DATE}}";

  // Extract headings for TOC
  const headings = [];
  const lines = content.split('\\n');
  lines.forEach((line, index) => {
    const match = line.match(/^(#{1,3})\\s+(.+)$/);
    if (match) {
      headings.push({
        id: \`heading-\${index}\`,
        text: match[2],
        level: match[1].length
      });
    }
  });

  // Simple markdown parser
  const parseMarkdown = (text) => {
    if (!text) return '';

    // Filter by search query
    if (searchQuery) {
      const filtered = text.split('\\n')
        .filter(line => line.toLowerCase().includes(searchQuery.toLowerCase()))
        .join('\\n');
      text = filtered;
    }

    return text.split('\\n').map((line, i) => {
      // Headers
      if (line.startsWith('### ')) {
        return <h3 key={i} style={styles.h3}>{line.slice(4)}</h3>;
      }
      if (line.startsWith('## ')) {
        return <h2 key={i} style={styles.h2}>{line.slice(3)}</h2>;
      }
      if (line.startsWith('# ')) {
        return <h1 key={i} style={styles.h1}>{line.slice(2)}</h1>;
      }

      // Lists
      if (line.startsWith('- ') || line.startsWith('* ')) {
        return <li key={i} style={styles.li}>{line.slice(2)}</li>;
      }
      if (line.match(/^\\d+\\. /)) {
        return <li key={i} style={styles.li}>{line.replace(/^\\d+\\. /, '')}</li>;
      }

      // Blockquote
      if (line.startsWith('> ')) {
        return <blockquote key={i} style={styles.blockquote}>{line.slice(2)}</blockquote>;
      }

      // Code blocks (simplified)
      if (line.startsWith(\`\\\`\\\`\`)) {
        return <div key={i} style={styles.codeBlock}>{line.slice(3)}</div>;
      }

      // Empty line
      if (line.trim() === '') {
        return <br key={i} />;
      }

      // Regular paragraph
      return <p key={i} style={styles.p}>{line}</p>;
    });
  };

  const scrollToTop = () => {
    window.scrollTo({ top: 0, behavior: 'smooth' });
  };

  return (
    <div style={styles.container}>
      {/* Header */}
      <header style={styles.header}>
        <div>
          <h1 style={styles.headerTitle}>{title}</h1>
          {subtitle && <p style={styles.headerSubtitle}>{subtitle}</p>}
        </div>
        <div style={styles.buttonGroup}>
          <button
            onClick={() => setShowToc(!showToc)}
            style={styles.button}
          >
            {showToc ? '📖 Hide TOC' : '📖 Show TOC'}
          </button>
          <button
            onClick={() => window.print()}
            style={styles.button}
          >
            📥 PDF
          </button>
        </div>
      </header>

      <div style={styles.mainLayout}>
        {/* Table of Contents */}
        {showToc && (
          <aside style={styles.sidebar}>
            <h2 style={styles.tocTitle}>📑 Table of Contents</h2>
            {headings.length === 0 ? (
              <p style={styles.noHeadings}>No headings found</p>
            ) : (
              <nav>
                {headings.map((heading) => (
                  <a
                    key={heading.id}
                    href={\`#\${heading.id}\`}
                    style={{
                      ...styles.tocLink,
                      paddingLeft: \`\${(heading.level - 1) * 16}px\`,
                      fontSize: heading.level === 1 ? '0.9rem' : '0.85rem',
                    }}
                  >
                    {heading.text}
                  </a>
                ))}
              </nav>
            )}
          </aside>
        )}

        {/* Main Content */}
        <main style={styles.main}>
          {/* Search */}
          <div style={styles.searchContainer}>
            <input
              type="text"
              placeholder="🔍 Search whitepaper..."
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              style={styles.searchInput}
            />
          </div>

          {/* Metadata */}
          {(author || date) && (
            <div style={styles.metadata}>
              {author && <div>✍️ <strong>Author:</strong> {author}</div>}
              {date && <div>📅 <strong>Date:</strong> {date}</div>}
            </div>
          )}

          {/* Content */}
          <article style={styles.article}>
            {parseMarkdown(content)}
          </article>
        </main>
      </div>

      {/* Back to Top Button */}
      <button
        onClick={scrollToTop}
        style={styles.backToTop}
      >
        ↑
      </button>
    </div>
  );
}
